%% --- Композиция: Quotient_Number содержит два Int_Number ---
Quotient_Number "1" *-- "1" Int_Number : up
Quotient_Number "1" *-- "1" Int_Number : doun

%% --- Использование: Int_Number использует NTT через Multiplication_NTT_Outward_Bypass ---
Int_Number ..> NTT : uses for multiplication
Int_Number ..> Error_Size_Array : throws on size mismatch
Int_Number ..> Error_Too_High_Degree : throws on degree overflow

%% --- Использование: Quotient_Number использует Int_Number ---
Quotient_Number ..> Int_Number : Uses

%% --- Внешние зависимости ---
Length_Int ..> Int_Number : used for length calculation
NTT ..> Error_Size_Array : throws if array size != 2^POWER_TWO
NTT ..> Error_Too_High_Degree : thrown via DispatchNTT default case
